> **给执行者：** 用户未授权子代理，按 `executing-plans.md` 串行执行。每项任务均使用复选框跟踪，并严格执行“失败测试 → 最小实现 → 定向验证 → 全量验证”。项目不是 Git 仓库，不执行提交、推送或分支操作。

**目标：** 修复已实证和后续审查发现的致命缺陷，落地统一候选、评分与分流中心，使成坨做账、多字段文字、重要性阈值、自然月差异池、可选大模型、详细 Excel 和整洁 GUI 形成一套可追溯的自动核对流程。

**架构：** 保留现有金额精度、索引、组合求和和 Excel 美化能力；匹配阶段只生成候选，由新的纯业务规则模块统一计算金额、评分、冲突和状态，再由报告层展示。大模型使用独立可选模块，只增强文字证据，失败时完整退回本地逻辑。

**技术栈：** Windows、Python 3.14、pandas、openpyxl、CustomTkinter、pytest、Python 标准库 `urllib`/`json`/`http.server`。

```python
def std_df(rows, source: str = "bank") -> pd.DataFrame:
    records = []
    for index, (date, amount, fields) in enumerate(rows):
        amount_decimal = Decimal(str(amount))
        text_fields = {str(key): str(value) for key, value in fields.items()}
        records.append({
            "date": pd.Timestamp(date),
            "amount": amount_decimal,
            "amount_decimal": PrecisionEngine.to_integer_li(amount_decimal),
            "summary": text_fields.get("摘要", ""),
            "aux_text_fields": text_fields,
            "balance": None,
            "voucher_no": "",
            "source": source,
            "original_idx": index + 1,
            "original_file_row": index + 2,
        })
    return pd.DataFrame(records)
```

```python
def trivial_candidate(date: str, **amounts: int) -> MatchCandidate:
    bank_income = amounts.get("bank_income", 0)
    journal_income = amounts.get("journal_income", 0)
    bank_expense = amounts.get("bank_expense", 0)
    journal_expense = amounts.get("journal_expense", 0)
    bank_gross = bank_income + bank_expense
    journal_gross = journal_income + journal_expense
    metrics = GroupMetrics(
        bank_gross_li=PrecisionEngine.to_integer_li(bank_gross),
        journal_gross_li=PrecisionEngine.to_integer_li(journal_gross),
        group_amount_li=PrecisionEngine.to_integer_li(max(bank_gross, journal_gross)),
        bank_income_li=PrecisionEngine.to_integer_li(bank_income),
        journal_income_li=PrecisionEngine.to_integer_li(journal_income),
        bank_expense_li=PrecisionEngine.to_integer_li(bank_expense),
        journal_expense_li=PrecisionEngine.to_integer_li(journal_expense),
        income_diff_li=PrecisionEngine.to_integer_li(abs(bank_income - journal_income)),
        expense_diff_li=PrecisionEngine.to_integer_li(abs(bank_expense - journal_expense)),
        total_diff_li=PrecisionEngine.to_integer_li(
            abs(bank_income - journal_income) + abs(bank_expense - journal_expense)
        ),
    )
    item = MatchCandidate(
        candidate_id=f"C-{date}-{len(amounts)}",
        bank_idxs=(0,),
        journal_idxs=(0,),
        match_type="tolerance_date",
        match_stage="测试",
        metrics=metrics,
        processing_status=ProcessingStatus.AUTO_CONFIRMED,
    )
    item.bank_dates = (pd.Timestamp(date),)
    item.journal_dates = (pd.Timestamp(date),)
    return item


def monthly_trivial_candidates(count: int, each_amount: int) -> list[MatchCandidate]:
    return [
        trivial_candidate(
            f"2026-01-{(index % 28) + 1:02d}",
            bank_income=each_amount,
            journal_income=0,
        )
        for index in range(count)
```

```python
def build_group_metrics(bank_amounts_li, journal_amounts_li):
    bank_income = sum(value for value in bank_amounts_li if value > 0)
    journal_income = sum(value for value in journal_amounts_li if value > 0)
    bank_expense = sum(abs(value) for value in bank_amounts_li if value < 0)
    journal_expense = sum(abs(value) for value in journal_amounts_li if value < 0)
    bank_gross = sum(abs(value) for value in bank_amounts_li)
    journal_gross = sum(abs(value) for value in journal_amounts_li)
    return GroupMetrics(
        bank_gross_li=bank_gross,
        journal_gross_li=journal_gross,
        group_amount_li=max(bank_gross, journal_gross),
        bank_income_li=bank_income,
        journal_income_li=journal_income,
        bank_expense_li=bank_expense,
        journal_expense_li=journal_expense,
        income_diff_li=abs(bank_income - journal_income),
        expense_diff_li=abs(bank_expense - journal_expense),
        total_diff_li=abs(bank_income - journal_income) + abs(bank_expense - journal_expense),
```

```python
def merge_labeled_fields(rows):
    merged: dict[str, str] = {}
    for row_number, fields in enumerate(rows, start=1):
        for field_name, value in fields.items():
            if str(value).strip():
                merged[f"第{row_number}笔·{field_name}"] = str(value)
    return merged


def _bigrams(text: str) -> set[str]:
    normalized = normalize_summary(text).upper()
    return {
        normalized[index:index + 2]
        for index in range(max(0, len(normalized) - 1))
    }


def score_text_fields(bank_fields, journal_fields):
    pair_scores: dict[str, float] = {}
    supporting: list[str] = []
    conflicting: list[str] = []
    sensitive_names = ("账号", "帐号", "卡号", "户名")

    for bank_name, bank_value in bank_fields.items():
        bank_norm = normalize_summary(bank_value)
        if not bank_norm:
            continue
        for journal_name, journal_value in journal_fields.items():
            journal_norm = normalize_summary(journal_value)
            if not journal_norm:
                continue
            bank_tokens = _bigrams(bank_norm)
            journal_tokens = _bigrams(journal_norm)
            union = bank_tokens | journal_tokens
            overlap = len(bank_tokens & journal_tokens) / len(union) if union else 0.0
            literal = calculate_similarity(bank_norm, journal_norm)
            pair_score = 1.0 if bank_norm == journal_norm else max(overlap, literal)
            label = f"{bank_name}↔{journal_name}"
            pair_scores[label] = pair_score
            if pair_score >= 0.5:
                supporting.append(f"{label}:{pair_score:.2f}")
            if (
                any(word in bank_name for word in sensitive_names)
                and any(word in journal_name for word in sensitive_names)
                and bank_norm != journal_norm
            ):
                conflicting.append(f"{label}:字段值不一致")

    top_scores = sorted(pair_scores.values(), reverse=True)[:3]
    if not top_scores:
        final_score = 0
    else:
        combined = top_scores[0] * 0.6 + (sum(top_scores) / len(top_scores)) * 0.4
        final_score = round(combined * 30)
    return TextEvidence(
        score=max(0, min(30, final_score)),
        field_scores=pair_scores,
        supporting_evidence=tuple(supporting),
        conflicting_evidence=tuple(conflicting),
    )
```

## 主要风险与控制

- **组合候选数量膨胀：** 保留日期窗口、候选数上限、DFS 深度、折半枚举和内存阈值；大模型只看本地预筛后的少量候选。
- **旧测试依赖“找到即占用”：** 保留旧对外字段并一次性回填，先用兼容层过渡，不让报告直接读取半成品候选。
- **明显微小错报自动处理被误解为精确匹配：** 报告分开披露精确匹配率、自动处理率和明显微小错报。
- **整池超限造成重复人工复核：** 待人工复核 Sheet 只放一条池级事项，组成明细只用于查看。
- **大模型泄密或中断主流程：** 在线字段白名单、密钥内存保存、日志脱敏、有限重试、超时后本地降级。
- **GUI 复杂度挤压日志：** 主窗口只显示摘要和入口，列映射、大模型、高级参数使用独立可缩放窗口。
- **无 Git 难以回滚：** 依赖已经验证的项目外完整备份；不执行覆盖式恢复或删除。
