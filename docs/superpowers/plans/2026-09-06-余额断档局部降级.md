**Goal:** 余额连续性断档只降级断档窗口（断档日＋前一有记录交易日）内的匹配关系，窗口外已精确对平的关系恢复自动确认。

def _流水(行):
    return pd.DataFrame(行, columns=["date", "amount", "balance", "original_file_row"])


def test_断档异常携带窗口日期():
    frame = _流水([
        ("2026-01-05", Decimal("5000"), Decimal("5000"), 2),
        ("2026-01-06", Decimal("100"), Decimal("4100"), 3),  # 开盘倒推4000，与上日末5000差1000
        ("2026-01-07", Decimal("50"), Decimal("4150"), 4),
    ])
    frame["date"] = pd.to_datetime(frame["date"])
    异常 = check_balance_continuity(frame, source="银行流水")

    assert str(异常[0]["窗口止"])[:10] == "2026-01-06"
    assert str(异常[0]["窗口起"])[:10] == "2026-01-05"


def test_断档窗口覆盖过半时回退整体降级():

    银行["date"] = pd.to_datetime(银行["date"])

    控制 = build_overall_controls(银行, 账)
