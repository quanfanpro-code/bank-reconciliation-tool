在 `Matcher` 内新增生成器：

```python
def _continuous_summary_groups(self, frame):
    if "original_idx" not in frame.columns:
        return
    current = []
    current_key = ""
    previous_row = None
    for index, row in frame.sort_values("original_idx").iterrows():
        key = normalize_summary(row.get("summary", ""))
        source_row = pd.to_numeric(row.get("original_idx"), errors="coerce")
        if pd.isna(source_row):
            if len(current) >= 2:
                yield current_key, current
            current, current_key, previous_row = [], "", None
            continue
        source_row = int(source_row)
        if key and key == current_key and previous_row is not None and source_row == previous_row + 1:
            current.append(int(index))
        else:
            if len(current) >= 2:
                yield current_key, current
            current = [int(index)] if key else []
        current_key = key
        previous_row = source_row if key else None
    if len(current) >= 2:
        yield current_key, current
```
